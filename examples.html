math-exercise – Overview of All Check Modes

Your answers are saved only in this browser.

This page briefly introduces and lets you try out every check mode of the math-exercise extension. See the README for full option details.

mode: equivalent (default)

Accepts any mathematically equivalent representation.

equivalent
Compute the area of a circle with r = 3 cm.
A = cm²

function binnedSamples(samples, count) {
  var bins = Array.from({ length: count }, function () { return []; });
  samples.forEach(function (point) {
    if (point[0] < 0 || point[0] > 1 || !Number.isFinite(point[1])) return;
    var index = Math.max(0, Math.min(count - 1, Math.round(point[0] * (count - 1))));
    bins[index].push(point[1]);
  });
  return bins.map(function (values, index) {
    if (!values.length) return null;
    var mean = values.reduce(function (sum, value) { return sum + value; }, 0) / values.length;
    return [index / (count - 1), mean];

}

var fittedGraph = null;

var controlPoints = [];
var controlGuides = [];
var regression = null;
var regressionDegree = 5;
var controlPointCount = 9;

function binnedSamples(samples, count) {
  var bins = Array.from({ length: count }, function () { return []; });
  samples.forEach(function (point) {
    if (point[0] < 0 || point[0] > 1 || !Number.isFinite(point[1])) return;
    var index = Math.max(0, Math.min(count - 1, Math.round(point[0] * (count - 1))));
    bins[index].push(point[1]);
  });
  return bins.map(function (values, index) {
    if (!values.length) return null;
    var mean = values.reduce(function (sum, value) { return sum + value; }, 0) / values.length;
    return [index / (count - 1), mean];
  }).filter(Boolean);
}

function removeControlPoints() {
  controlPoints.forEach(function (point) { board.removeObject(point); });
  controlGuides.forEach(function (guide) { board.removeObject(guide); });
  controlPoints = [];
  controlGuides = [];
}

function updateInterpolation(activePoint) {
  regression = null;
  if (controlPoints.length < regressionDegree + 1) return;
  var dataX = controlPoints.map(function (point) { return point.X(); });
  var dataY = controlPoints.map(function (point) { return point.Y(); });
  var fitted = JXG.Math.Numerics.regressionPolynomial(regressionDegree, dataX, dataY);
  var offset = activePoint ? activePoint.Y() - fitted(activePoint.X()) : 0;

  controlPoints.forEach(function (point, index) {
    if (point !== activePoint) {
      point.setPosition(JXG.COORDS_BY_USER, [dataX[index], fitted(dataX[index]) + offset]);
    }
  });
  var projectedY = controlPoints.map(function (point) { return point.Y(); });
  regression = JXG.Math.Numerics.regressionPolynomial(regressionDegree, dataX, projectedY);

  if (!fittedGraph) {
    fittedGraph = board.create('functiongraph', [function (x) {
      return regression ? regression(x) : NaN;
    }, 0, 1], {
      strokeColor: '#dc2626',
      strokeWidth: 4,
      fixed: true
    });
  }
  board.fullUpdate();
}

function createControlPoints() {
  var samples = rawSamples.filter(function (point) {
    return point[0] >= 0 && point[0] <= 1 && Number.isFinite(point[1]);
  });
  if (samples.length < regressionDegree + 1) return;

  var coords = samples.map(function (point) {
    return new JXG.Coords(JXG.COORDS_BY_USER, point, board);
  });
  var simplified = JXG.Math.Numerics.Visvalingam(coords, controlPointCount - 2);
  controlPoints = simplified.map(function (coord) {
    var x = coord.usrCoords[1];
    var guide = board.create('line', [[x, 0], [x, 1]], {
      visible: false,
      fixed: true
    });
    controlGuides.push(guide);
    var point = board.create('glider', [x, coord.usrCoords[2], guide], {
      name: '',
      size: 4,
      strokeColor: '#1d4ed8',
      fillColor: '#60a5fa',
      highlightStrokeColor: '#1e40af',
      highlightFillColor: '#93c5fd'
    });
    point.on('drag', function () { updateInterpolation(point); });
    point.on('up', function () { updateInterpolation(point); });
    return point;
  });

  updateInterpolation();
}

board.on('down', function (event) {
  if (drawing) return;
  var hits = board.getAllObjectsUnderMouse ? board.getAllObjectsUnderMouse(event) : [];
  if (hits.some(function (object) { return object.elType === 'point' || object.elType === 'glider'; })) return;
  drawing = true;
  rawSamples = [];
  regression = null;
  if (sketch) board.removeObject(sketch);
  if (fittedGraph) board.removeObject(fittedGraph);
  fittedGraph = null;
  removeControlPoints();
  sketch = board.create('curve', [[], []], {
    strokeColor: '#94a3b8',
    strokeWidth: 7,
    lineCap: 'round',
    fixed: true
  });
  if (event.pointerId !== undefined && board.containerObj.setPointerCapture) {
    board.containerObj.setPointerCapture(event.pointerId);
  }
});

board.on('move', function (event) {
  if (!drawing) return;
  var screen = board.getMousePosition(event);
  var coords = new JXG.Coords(JXG.COORDS_BY_SCREEN, screen, board);
  var point = [coords.usrCoords[1], coords.usrCoords[2]];
  rawSamples.push(point);
  sketch.dataX.push(point[0]);
  sketch.dataY.push(point[1]);
  board.update();
});

function finishEditableSketch(event) {
  if (!drawing) {
    board.mode = board.BOARD_MODE_NONE;
    return;
  }
  drawing = false;
  board.mode = board.BOARD_MODE_NONE;
  if (event && event.pointerId !== undefined && board.containerObj.hasPointerCapture &&
      board.containerObj.hasPointerCapture(event.pointerId)) {
    board.containerObj.releasePointerCapture(event.pointerId);
  }
  createControlPoints();
}

board.on('up', finishEditableSketch);
board.containerObj.addEventListener('pointercancel', finishEditableSketch);
board.containerObj.addEventListener('lostpointercapture', finishEditableSketch);
window.addEventListener('blur', finishEditableSketch);

function responseData() {
  var inside = rawSamples.filter(function (point) { return point[0] >= 0 && point[0] <= 1; });
  var xs = inside.map(function (point) { return point[0]; });
  var coverage = xs.length ? Math.max.apply(null, xs) - Math.min.apply(null, xs) : 0;
  var fitSamples = [];
  if (regression) {
    for (var i = 0; i <= 200; i++) {
      var x = i / 200;
      fitSamples.push([x, regression(x)]);
    }
  }
  return {
    representation: 'freehand-polynomial-regression-controls',
    interval: [0, 1],
    rawSamples: rawSamples,
    controlPoints: controlPoints.map(function (point) { return [point.X(), point.Y()]; }),
    coverage: coverage,
    regression: regression ? {
      degree: regressionDegree,
      polynomialValues: Array.from({ length: 6 }, function (_, i) { return regression(i / 5); }),
      term: regression.getTerm ? regression.getTerm() : '',
      samples: fitSamples
    } : null
  };
}

Mini-machine number representation

The mini-machine uses the simplified floating-point format

\[ \boxed{S}\quad\boxed{EEE}\quad\boxed{MMMM}, \]

where:

  • \(S\) is one sign bit;
  • \(E\) is a three-bit exponent field;
  • \(M\) is a four-bit mantissa field;
  • the exponent bias is \(3\).

Only normalized numbers with \(E=1,2,\ldots,6\) are used. The bit patterns \(E=0\) and \(E=7\) are excluded, so this mini-machine ignores subnormal numbers, infinities, and NaNs.

A bit pattern represents the number

\[ (-1)^S\left(1+\frac{M}{16}\right)2^{E-3}, \]

where \(M\) is the integer value of the four mantissa bits.

For example,

\[ 0\;|\;100\;|\;0110 \]

has \(S=0\), \(E=(100)_2=4\), \(e=4-3=1\), and \(M=(0110)_2=6\). Therefore,

\[ x=(1+6/16)2^1=2.75. \]

To encode a representable number, first write it as

\[ \pm(1.MMMM)_2\,2^e. \]

Then determine the sign bit, calculate the stored exponent \(E=e+3\), and write \(S\), \(E\), and \(M\) as bit fields.
