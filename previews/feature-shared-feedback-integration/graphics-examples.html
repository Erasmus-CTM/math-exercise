Graphical answers with JSXGraph

These exercises are answered directly in the plots. After placing points, moving handles, or drawing a curve, open the exercise heading below the plot and select Check. The collapsed source panels show authors how each JSXGraph plot and its answer checker are defined.

Click every local maximum

A local maximum is a point whose function value is at least as large as the values at all sufficiently nearby points—a visible crest of the graph. One of four sinusoidal functions is selected when the page loads. Click once at every visible local maximum; each answer appears as an orange point. Use Clear clicks to remove all selected points or Show another function to try the next variant.

Show JSXGraph source and policy YAML
```{.jsxgraph assessment_id="pooled-maxima-board" width="900" height="430" style="border:1px solid #ddd; border-radius:8px; max-width:100%;"}
var xMax = 8 * Math.PI;
var variants = [
  { amplitude: 1.00, phase: 0.00, offset: 0.00, color: '#2563eb' },
  { amplitude: 0.75, phase: 0.55, offset: 0.20, color: '#dc2626' },
  { amplitude: 1.10, phase: -0.70, offset: -0.10, color: '#059669' },
  { amplitude: 0.85, phase: 1.05, offset: 0.10, color: '#7c3aed' }
];
var variantIndex = Math.floor(Math.random() * variants.length);

var board = JXG.JSXGraph.initBoard(BOARDID, {
  boundingbox: [-0.8, 1.9, xMax + 0.8, -1.9],
  axis: true,
  pan: { enabled: false },
  zoom: { enabled: false },
  showNavigation: false
});

var graph = null;
var selectedPoints = [];
var variantText = board.create('text', [0.6, 1.55, ''], {
  fixed: true,
  fontSize: 16,
  color: '#475569'
});

function currentFunction(x) {
  var cfg = variants[variantIndex];
  return cfg.offset + cfg.amplitude * Math.sin(x + cfg.phase);
}

function clearClicks() {
  selectedPoints.forEach(function (point) { board.removeObject(point); });
  selectedPoints = [];
  board.update();
}

function drawVariant() {
  clearClicks();
  if (graph) board.removeObject(graph);
  var cfg = variants[variantIndex];
  graph = board.create('functiongraph', [currentFunction, 0, xMax], {
    strokeColor: cfg.color,
    strokeWidth: 3,
    fixed: true,
    highlight: false
  });
  variantText.setText('Function ' + (variantIndex + 1) + ' of ' + variants.length);
  board.update();
}

function nextVariant() {
  variantIndex = (variantIndex + 1) % variants.length;
  drawVariant();
}

board.create('button', [0.6, -1.55, 'Show another function', nextVariant], {
  fixed: true
});
board.create('button', [5.3, -1.55, 'Clear clicks', clearClicks], {
  fixed: true
});

board.on('down', function (event) {
  if (event.target && event.target.closest && event.target.closest('button')) return;
  var screen = board.getMousePosition(event);
  var coords = new JXG.Coords(JXG.COORDS_BY_SCREEN, screen, board);
  var x = coords.usrCoords[1];
  var y = coords.usrCoords[2];
  if (x < 0 || x > xMax || y < -1.35 || y > 1.35) return;
  selectedPoints.push(board.create('point', [x, y], {
    name: '',
    size: 4,
    fixed: true,
    color: '#f59e0b',
    strokeColor: '#92400e',
    showInfobox: false
  }));
});

drawVariant();

JXG.QuartoAssessment.register({
  board: board,
  response: function () {
    return {
      representation: 'clicked-local-maxima',
      variant: variantIndex,
      clicks: selectedPoints.map(function (point) { return [point.X(), point.Y()]; })
    };
  },
  ai: {
    render: true,
    summary: function (data) {
      return {
        representation: data.representation,
        functionVariant: data.variant + 1,
        selectedPointCount: data.clicks.length,
        feedbackPolicy: 'Discuss how local maxima are identified visually. Do not reveal target coordinates.'
      };
    }
  }
});
```

Mark exactly one period

A positive number \(T\) is a period of a function \(f\) when \(f(x+T)=f(x)\) for every \(x\); one complete period runs between two corresponding points in consecutive repetitions of the graph. The graph below shows five periods. Move the two orange dots labelled L and R horizontally along the line below the graph. They control the left and right edges of the shaded frame. Your answer is accepted anywhere on the graph when the frame spans exactly one complete period. Do not draw on the curve; only move the two orange dots.

Show JSXGraph source and policy YAML
```{.jsxgraph assessment_id="period-window-board" width="900" height="430" style="border:1px solid #ddd; border-radius:8px; max-width:100%;"}
var truePeriod = 2.5;
var plotEnd = 5 * truePeriod;
var yLow = -1.25;
var yHigh = 1.25;

  boundingbox: [-0.5, 1.85, plotEnd + 0.5, -1.85],

  pan: { enabled: false },
  zoom: { enabled: false },
  showNavigation: false

  return 0.85 * Math.sin(2 * Math.PI * x / truePeriod + 0.35)

}

board.create('functiongraph', [periodicFunction, 0, plotEnd], {
  strokeColor: '#2563eb',
  strokeWidth: 3,
  fixed: true,
  highlight: false

var handleGuide = board.create('line', [[0, -1.48], [1, -1.48]], {
  visible: false,

  name: 'L',

var rightHandle = board.create('glider', [4.8, -1.48, handleGuide], {
  name: 'R',
  size: 6,
  color: '#f59e0b',
  showInfobox: false

function leftX() { return Math.max(0, Math.min(leftHandle.X(), rightHandle.X())); }
function rightX() { return Math.min(plotEnd, Math.max(leftHandle.X(), rightHandle.X())); }

var corners = [
```

The next three examples show other ways to answer graphically: moving fixed points, drawing a curve in one stroke, and refining a drawn curve with control points.

Construct a non-decreasing piecewise-linear function

Drag the five blue points vertically. Consecutive points are joined by straight segments. Flat segments are allowed, but no segment may decrease from left to right, and the final point must be higher than the first.

Show JSXGraph source and policy YAML
```{.jsxgraph assessment_id="pwl-monotonicity-board" width="800" height="420" style="border:1px solid #ddd; border-radius:8px; max-width:100%;"}
var board = JXG.JSXGraph.initBoard(BOARDID, {
  boundingbox: [-0.1, 1.25, 1.1, -0.25],
  axis: true,
  pan: { enabled: false },
  zoom: { enabled: false },
  showNavigation: false
});

var xs = [0, 0.25, 0.5, 0.75, 1];
var points = xs.map(function (x, index) {
  var guide = board.create('line', [[x, 0], [x, 1]], {
    visible: false,
    fixed: true
  });
  return board.create('glider', [x, 0.35, guide], {
    name: 'P_' + index,
    size: 5,
    color: '#2563eb',
    showInfobox: false
  });
});

for (var i = 0; i < points.length - 1; i++) {
  board.create('segment', [points[i], points[i + 1]], {
    strokeWidth: 3,
    strokeColor: '#2563eb',
    fixed: true
  });
}

function pointData() {
  return points.map(function (point) { return [point.X(), point.Y()]; });
}

JXG.QuartoAssessment.register({
  board: board,
  response: function () {
    return {
      representation: 'piecewise-linear',
      interval: [0, 1],
      points: pointData()
    };
  },
  ai: {
    render: true,
    summary: function (data) {
      var slopes = data.points.slice(1).map(function (point, i) {
        return (point[1] - data.points[i][1]) / (point[0] - data.points[i][0]);
      });
      return {
        representation: data.representation,
        interval: data.interval,
        segmentCount: slopes.length,
        positiveSegments: slopes.filter(function (s) { return s > 1e-8; }).length,
        flatSegments: slopes.filter(function (s) { return Math.abs(s) <= 1e-8; }).length,
        decreasingSegments: slopes.filter(function (s) { return s < -1e-8; }).length,
        minimumSlope: Math.min.apply(null, slopes)
      };
    }
  }
});
```

Draw a non-decreasing curve

Draw one continuous stroke from near \(x=0\) to near \(x=1\). A grey line follows your pointer, and a smooth red curve appears when you release it. The red curve is the submitted answer: it must cover almost the whole interval, never decrease, and finish above where it starts. Flat portions are allowed.

Show JSXGraph source and policy YAML
```{.jsxgraph assessment_id="freehand-monotonicity-board" width="800" height="440" style="border:1px solid #ddd; border-radius:8px; max-width:100%;"}
var board = JXG.JSXGraph.initBoard(BOARDID, {
  boundingbox: [-0.1, 1.25, 1.1, -0.25],
  axis: true,
  pan: { enabled: false },
  zoom: { enabled: false },
  showNavigation: false
});

var drawing = false;
var sketch = null;
var fittedGraph = null;
var rawSamples = [];
var regression = null;
var regressionDegree = 5;

function binnedSamples(samples, count) {
  var bins = Array.from({ length: count }, function () { return []; });
  samples.forEach(function (point) {
    if (point[0] < 0 || point[0] > 1 || !Number.isFinite(point[1])) return;
    var index = Math.max(0, Math.min(count - 1, Math.round(point[0] * (count - 1))));
    bins[index].push(point[1]);
  });
  return bins.map(function (values, index) {
    if (!values.length) return null;
    var mean = values.reduce(function (sum, value) { return sum + value; }, 0) / values.length;
    return [index / (count - 1), mean];
  }).filter(Boolean);
}

function fitSketch() {
  var samples = binnedSamples(rawSamples, 51);
  regression = null;
  if (fittedGraph) board.removeObject(fittedGraph);
  fittedGraph = null;
  if (samples.length < regressionDegree + 1) return;

  var dataX = samples.map(function (point) { return point[0]; });
  var dataY = samples.map(function (point) { return point[1]; });
  regression = JXG.Math.Numerics.regressionPolynomial(regressionDegree, dataX, dataY);
  fittedGraph = board.create('functiongraph', [regression, 0, 1], {
    strokeColor: '#dc2626',
    strokeWidth: 4,
    fixed: true
  });
  board.update();
}

board.on('down', function (event) {
  if (drawing) return;
  drawing = true;
  rawSamples = [];
  regression = null;
  if (sketch) board.removeObject(sketch);
  if (fittedGraph) board.removeObject(fittedGraph);
  fittedGraph = null;
  sketch = board.create('curve', [[], []], {
    strokeColor: '#94a3b8',
    strokeWidth: 7,
    lineCap: 'round',
    fixed: true
  });
  if (event.pointerId !== undefined && board.containerObj.setPointerCapture) {
    board.containerObj.setPointerCapture(event.pointerId);
  }
});

board.on('move', function (event) {
  if (!drawing) return;
  var screen = board.getMousePosition(event);
  var coords = new JXG.Coords(JXG.COORDS_BY_SCREEN, screen, board);
  var point = [coords.usrCoords[1], coords.usrCoords[2]];
  rawSamples.push(point);
  sketch.dataX.push(point[0]);
  sketch.dataY.push(point[1]);
  board.update();
});

function finishSketch(event) {
  if (!drawing) {
    board.mode = board.BOARD_MODE_NONE;
    return;
  }
  drawing = false;
  board.mode = board.BOARD_MODE_NONE;
  if (event && event.pointerId !== undefined && board.containerObj.hasPointerCapture &&
      board.containerObj.hasPointerCapture(event.pointerId)) {
    board.containerObj.releasePointerCapture(event.pointerId);
  }
  fitSketch();
}

board.on('up', finishSketch);
board.containerObj.addEventListener('pointercancel', finishSketch);
board.containerObj.addEventListener('lostpointercapture', finishSketch);
window.addEventListener('blur', finishSketch);

function responseData() {
  var inside = rawSamples.filter(function (point) { return point[0] >= 0 && point[0] <= 1; });
  var xs = inside.map(function (point) { return point[0]; });
  var coverage = xs.length ? Math.max.apply(null, xs) - Math.min.apply(null, xs) : 0;
  var fitSamples = [];
  if (regression) {
    for (var i = 0; i <= 200; i++) {
      var x = i / 200;
      fitSamples.push([x, regression(x)]);
    }
  }
  return {
    representation: 'freehand-polynomial-regression',
    interval: [0, 1],
    rawSamples: rawSamples,
    coverage: coverage,
    regression: regression ? {
      degree: regressionDegree,
      term: regression.getTerm ? regression.getTerm() : '',
      samples: fitSamples
    } : null
  };
}

JXG.QuartoAssessment.register({
  board: board,
  response: responseData,
  ai: {
    render: true,
    summary: function (data) {
      var samples = data.regression ? data.regression.samples : [];
      var slopes = samples.slice(1).map(function (point, i) {
        return (point[1] - samples[i][1]) / (point[0] - samples[i][0]);
      });
      var decreasingIntervals = [];
      var decreasingStart = null;
      slopes.forEach(function (slope, i) {
        if (slope < -1e-3 && decreasingStart === null) decreasingStart = samples[i][0];
        if (slope >= -1e-3 && decreasingStart !== null) {
          decreasingIntervals.push([decreasingStart, samples[i][0]]);
          decreasingStart = null;
        }
      });
      if (decreasingStart !== null && samples.length) {
        decreasingIntervals.push([decreasingStart, samples[samples.length - 1][0]]);
      }
      return {
        representation: data.representation,
        interval: data.interval,
        rawSampleCount: data.rawSamples.length,
        domainCoverage: data.coverage,
        polynomialDegree: data.regression ? data.regression.degree : null,
        minimumSampledSlope: slopes.length ? Math.min.apply(null, slopes) : null,
        nondecreasingSlopeFraction: slopes.length ? slopes.filter(function (slope) { return slope >= -1e-3; }).length / slopes.length : null,
        decreasingIntervals: decreasingIntervals,
        feedbackPolicy: 'Discuss only the visible shape and qualitative behavior. Do not reveal, reconstruct, or display polynomial coefficients, fitted expressions, symbolic derivatives, raw samples, or point coordinates.'
      };
    }
  }
});
```

Draw and refine a non-decreasing curve

First draw one continuous stroke from near \(x=0\) to near \(x=1\). A smooth red curve and blue adjustment points then appear. Drag the blue points vertically to refine the red curve. The red curve is the submitted answer: it must cover almost the whole interval, never decrease, and rise overall.

```{.jsxgraph assessment_id="editable-freehand-monotonicity-board" width="800" height="440" style="border:1px solid #ddd; border-radius:8px; max-width:100%;"}
var controlGuides = [];

  samples.forEach(function (point) {
    if (point[0] < 0 || point[0] > 1 || !Number.isFinite(point[1])) return;
    var index = Math.max(0, Math.min(count - 1, Math.round(point[0] * (count - 1))));
    bins[index].push(point[1]);
  });
  return bins.map(function (values, index) {
    if (!values.length) return null;
    var mean = values.reduce(function (sum, value) { return sum + value; }, 0) / values.length;
    return [index / (count - 1), mean];
  }).filter(Boolean);
}

function removeControlPoints() {
  controlPoints.forEach(function (point) { board.removeObject(point); });
  controlGuides.forEach(function (guide) { board.removeObject(guide); });
  controlPoints = [];
  controlGuides = [];
}

function updateInterpolation(activePoint) {
  regression = null;
  if (controlPoints.length < regressionDegree + 1) return;
  var dataX = controlPoints.map(function (point) { return point.X(); });
  var dataY = controlPoints.map(function (point) { return point.Y(); });
  var fitted = JXG.Math.Numerics.regressionPolynomial(regressionDegree, dataX, dataY);
  var offset = activePoint ? activePoint.Y() - fitted(activePoint.X()) : 0;

  controlPoints.forEach(function (point, index) {
    if (point !== activePoint) {
      point.setPosition(JXG.COORDS_BY_USER, [dataX[index], fitted(dataX[index]) + offset]);
    }
  });
  var projectedY = controlPoints.map(function (point) { return point.Y(); });
  regression = JXG.Math.Numerics.regressionPolynomial(regressionDegree, dataX, projectedY);

  if (!fittedGraph) {
    fittedGraph = board.create('functiongraph', [function (x) {
      return regression ? regression(x) : NaN;
    }, 0, 1], {
      strokeColor: '#dc2626',
      strokeWidth: 4,
      fixed: true
    });
  }
  board.fullUpdate();
}

function createControlPoints() {
  var samples = rawSamples.filter(function (point) {
    return point[0] >= 0 && point[0] <= 1 && Number.isFinite(point[1]);
  });
  if (samples.length < regressionDegree + 1) return;

  var coords = samples.map(function (point) {
    return new JXG.Coords(JXG.COORDS_BY_USER, point, board);
  });
  var simplified = JXG.Math.Numerics.Visvalingam(coords, controlPointCount - 2);
  controlPoints = simplified.map(function (coord) {
    var x = coord.usrCoords[1];
    var guide = board.create('line', [[x, 0], [x, 1]], {
      visible: false,
      fixed: true
    });
    controlGuides.push(guide);
    var point = board.create('glider', [x, coord.usrCoords[2], guide], {
      name: '',
      size: 4,
      strokeColor: '#1d4ed8',
      fillColor: '#60a5fa',
      highlightStrokeColor: '#1e40af',
      highlightFillColor: '#93c5fd'
    });
    point.on('drag', function () { updateInterpolation(point); });
    point.on('up', function () { updateInterpolation(point); });
    return point;
  });

  updateInterpolation();
}
```

Coordinate transformations

The final activities connect parameter descriptions to geometry. In each one, a small set of algebraic parameters determines a visible object or region. Matching a target therefore requires reasoning in both representations: which parameter is wrong, and what geometric change will correcting it produce?

Transform a cuboid in 3D

An affine transformation combines a linear change of shape and orientation with a translation. In this activity the map has the form \(T(p)=sR_z(\theta)p+t\): the cuboid is scaled uniformly by \(s\), rotated about the \(z\)-axis by \(\theta\), and translated by \(t=(t_x,t_y,t_z)\). Matching the wireframes builds a geometric understanding of how those parameters affect a 3D object and how the operations compose.
