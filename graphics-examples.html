Click every local maximum

A local maximum is a point whose function value is at least as large as the values at all sufficiently nearby points. Here, mark the visible crests: interior points where the graph stops rising and starts falling. The ends of the plotted interval do not count, even where the graph rises into the right edge. One of four sinusoidal functions is selected when the page loads. Click once at every crest; each answer appears as an orange point. Use Clear clicks to remove all selected points or Show another function to try the next variant.

```{.jsxgraph assessment_id="pooled-maxima-board" width="900" height="430" style="border:1px solid #ddd; border-radius:8px; max-width:100%;"}
var xMax = 8 * Math.PI;
var variants = [
  { amplitude: 1.00, phase: 0.00, offset: 0.00, color: '#2563eb' },
  { amplitude: 0.75, phase: 0.55, offset: 0.20, color: '#dc2626' },
  { amplitude: 1.10, phase: -0.70, offset: -0.10, color: '#059669' },
  { amplitude: 0.85, phase: 1.05, offset: 0.10, color: '#7c3aed' }
];
var variantIndex = Math.floor(Math.random() * variants.length);

var board = JXG.JSXGraph.initBoard(BOARDID, {
  boundingbox: [-0.8, 1.9, xMax + 0.8, -1.9],
  axis: true,
  pan: { enabled: false },
  zoom: { enabled: false },
  showNavigation: false
});

var graph = null;
var selectedPoints = [];
var variantText = board.create('text', [0.6, 1.55, ''], {
  fixed: true,
  fontSize: 16,
  color: '#475569'
});

function currentFunction(x) {
  var cfg = variants[variantIndex];
  return cfg.offset + cfg.amplitude * Math.sin(x + cfg.phase);
}

function clearClicks() {
  selectedPoints.forEach(function (point) { board.removeObject(point); });
  selectedPoints = [];
  board.update();
}

function drawVariant() {
  clearClicks();
  if (graph) board.removeObject(graph);
  var cfg = variants[variantIndex];
  graph = board.create('functiongraph', [currentFunction, 0, xMax], {
    strokeColor: cfg.color,
    strokeWidth: 3,
    fixed: true,
    highlight: false
  });
  variantText.setText('Function ' + (variantIndex + 1) + ' of ' + variants.length);
  board.update();
}

function nextVariant() {
  variantIndex = (variantIndex + 1) % variants.length;
  drawVariant();
}

board.create('button', [0.6, -1.55, 'Show another function', nextVariant], {
  fixed: true
});
board.create('button', [5.3, -1.55, 'Clear clicks', clearClicks], {
  fixed: true
});

board.on('down', function (event) {
  if (event.target && event.target.closest && event.target.closest('button')) return;
  var screen = board.getMousePosition(event);
  var coords = new JXG.Coords(JXG.COORDS_BY_SCREEN, screen, board);
  var x = coords.usrCoords[1];
  var y = coords.usrCoords[2];
  if (x < 0 || x > xMax || y < -1.35 || y > 1.35) return;
  selectedPoints.push(board.create('point', [x, y], {
    name: '',
    size: 4,
    fixed: true,
    color: '#f59e0b',
    strokeColor: '#92400e',
    showInfobox: false
  }));
});

drawVariant();

JXG.QuartoAssessment.register({
  board: board,
  response: function () {
    return {
      representation: 'clicked-local-maxima',
      variant: variantIndex,
      clicks: selectedPoints.map(function (point) { return [point.X(), point.Y()]; })
    };
  },
  ai: {
    render: true,
    summary: function (data) {
      return {
        representation: data.representation,
        functionVariant: data.variant + 1,
        selectedPointCount: data.clicks.length,
        feedbackPolicy: 'Discuss how local maxima are identified visually. Do not reveal target coordinates.'
      };
    }
  }
});
```

Mark exactly one period

A positive number \(T\) is a period of a function \(f\) when \(f(x+T)=f(x)\) for every \(x\); one complete period runs between two corresponding points in consecutive repetitions of the graph. The graph below shows five periods. Move the two orange dots labelled L and R horizontally along the line below the graph. They control the left and right edges of the shaded frame. Your answer is accepted anywhere on the graph when the frame spans exactly one complete period. Do not draw on the curve; only move the two orange dots.

Construct a non-decreasing piecewise-linear function

Drag the five blue points vertically. Consecutive points are joined by straight segments. Flat segments are allowed, but no segment may decrease from left to right, and the final point must be higher than the first.

Show JSXGraph code
```{.jsxgraph assessment_id="pwl-monotonicity-board" width="800" height="420" style="border:1px solid #ddd; border-radius:8px; max-width:100%;"}
var board = JXG.JSXGraph.initBoard(BOARDID, {
  boundingbox: [-0.1, 1.25, 1.1, -0.25],
  axis: true,
  pan: { enabled: false },
  zoom: { enabled: false },
  showNavigation: false
});

var xs = [0, 0.25, 0.5, 0.75, 1];
var points = xs.map(function (x, index) {
  var guide = board.create('line', [[x, 0], [x, 1]], {
    visible: false,
    fixed: true
  });
  return board.create('glider', [x, 0.35, guide], {
    name: 'P_' + index,
    size: 5,
    color: '#2563eb',
    showInfobox: false
  });
});

for (var i = 0; i < points.length - 1; i++) {
  board.create('segment', [points[i], points[i + 1]], {
    strokeWidth: 3,
    strokeColor: '#2563eb',
    fixed: true
  });
}

function pointData() {
  return points.map(function (point) { return [point.X(), point.Y()]; });
}

JXG.QuartoAssessment.register({
  board: board,
  response: function () {
    return {
      representation: 'piecewise-linear',
      interval: [0, 1],
      points: pointData()
    };
  },
  ai: {
    render: true,
    summary: function (data) {
      var slopes = data.points.slice(1).map(function (point, i) {
        return (point[1] - data.points[i][1]) / (point[0] - data.points[i][0]);
      });
      return {
        representation: data.representation,
        interval: data.interval,
        segmentCount: slopes.length,
        positiveSegments: slopes.filter(function (s) { return s > 1e-8; }).length,
        flatSegments: slopes.filter(function (s) { return Math.abs(s) <= 1e-8; }).length,
        decreasingSegments: slopes.filter(function (s) { return s < -1e-8; }).length,
        minimumSlope: Math.min.apply(null, slopes)
      };
    }
  }
});
```

Coordinate transformations

The final activities connect parameter descriptions to geometry. In each one, a small set of algebraic parameters determines a visible object or region. Matching a target therefore requires reasoning in both representations: which parameter is wrong, and what geometric change will correcting it produce?

Transform a cuboid in 3D

An affine transformation combines a linear change of shape and orientation with a translation. In this activity the map has the form \(T(p)=sR_z(\theta)p+t\): the cuboid is scaled uniformly by \(s\), rotated about the \(z\)-axis by \(\theta\), and translated by \(t=(t_x,t_y,t_z)\). Matching the wireframes builds a geometric understanding of how those parameters affect a 3D object and how the operations compose.

Match the orange cuboid to the fixed green target. Move the five sliders for translation, uniform scale, and rotation about the \(z\)-axis. The unequal side lengths make the orientation visible. You may drag the 3D view to inspect the result; only the sliders are assessed.

Match a polar-coordinate region

A rectangle \([r_{\min},r_{\max}]\times[\varphi_{\min},\varphi_{\max}]\) in parameter space becomes an annular sector under \((r,\varphi)\mapsto(r\cos\varphi,r\sin\varphi)\). The blue target is an annular sector. Move the four sliders to match it with the orange outline. Radii are ordinary coordinates; the two angle sliders are multiples of \(\pi\) (for example, 0.5 means \(\pi/2\)). Only the slider values are assessed.
